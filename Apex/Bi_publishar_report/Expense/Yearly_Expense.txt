
select  ed.EXPENSE_M_ID,EXPENSE_TYPE_NAME,EXPENSE_AMOUNT,BRANCH_NAME,to_char(EXPENSE_DATE,'fmYYYY') EXPENSE_DATE
from    expense_m em,EXPENSE_D ed,EXPENSE_TYPE et,BRANCH b
where   ed.EXPENSE_M_ID = em.EXPENSE_M_ID
and     ed.EXPENSE_TYPE_id = et.EXPENSE_TYPE_id 
and     em.BRANCH_ID = b.BRANCH_ID
and       to_char(em.EXPENSE_DATE,'fmYYYY') =  :P63_YEAR;
