WITH
---------Total_Rooms Query --------
Total_Rooms
AS
(
select count(Room_id) Total_Rooms
from Room
)



---------Total_Salary Query --------
Total_Salary
AS
(
select sum(salary) Total_Salary
from employee
),
--------------------------------
---------Total_User Query --------
Total_User
AS
(
select count(username) Total_User
from user_info
)
--------------------------------
