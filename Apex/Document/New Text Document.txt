declare
  v_Room_Status_id Room_Status.Room_Status_id%type ;
begin
    select nvl(max(to_number(Room_Status_id)), 0) + 1
    into   v_Room_Status_id
    from   Room_Status ;
    pro_Room_Status_insert(v_Room_Status_id, :BOOKING_M.BOOKING_M_ID, :BOOKING_D.ROOM_ID, :BOOKING_M.CHECK_IN_DATE, :BOOKING_M.CHECK_OUT_DATE,:BOOKING_M.ACTION_USER) ;
end ;
