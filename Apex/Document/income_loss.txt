from     EXPENSE_M em, EXPENSE_D ed

Current Months Profit/Loss

SQL Query :


 select     case when  ( select   nvl(sum(pay), 0) 
            from     bill 
            where    to_char(bill_date, 'MM') = to_char(sysdate, 'MM') ) < (select   nvl(sum(EXPENSE_AMOUNT), 0)
            from     EXPENSE_M em, EXPENSE_D ed
            where    em.EXPENSE_M_ID = ed.EXPENSE_M_ID
            and      to_char(EXPENSE_DATE, 'MM') = to_char(sysdate, 'MM') ) then
            'Loss- '||((select   nvl(sum(EXPENSE_AMOUNT), 0)
            from     EXPENSE_M em, EXPENSE_D ed
            where    em.EXPENSE_M_ID = ed.EXPENSE_M_ID
            and      to_char(EXPENSE_DATE, 'MM') = to_char(sysdate, 'MM') ) -  ( select   nvl(sum(pay), 0) Total_income
             from     bill 
             where    to_char(bill_date, 'MM') = to_char(sysdate, 'MM') ))
            when     ( select   nvl(sum(pay), 0) 
             from     bill 
             where    to_char(bill_date, 'MM') = to_char(sysdate, 'MM') ) > (select   nvl(sum(EXPENSE_AMOUNT), 0)
            from     EXPENSE_M em, EXPENSE_D ed
            where    em.EXPENSE_M_ID = ed.EXPENSE_M_ID
            and      to_char(EXPENSE_DATE, 'MM') = to_char(sysdate, 'MM') ) then
            'Profit- '|| (( select   nvl(sum(pay), 0) 
             from     bill 
             where    to_char(bill_date, 'MM') = to_char(sysdate, 'MM') ) - (select   nvl(sum(EXPENSE_AMOUNT), 0)
            from     EXPENSE_M em, EXPENSE_D ed
            where    em.EXPENSE_M_ID = ed.EXPENSE_M_ID
            and      to_char(EXPENSE_DATE, 'MM') = to_char(sysdate, 'MM') ))
            else 'No profit'
            end as loss_Profit
 from    dual ;
