SELECT Booking_M.Guest_ID,BOOKING_ID,FIRST_NAME||' '||LAST_NAME Guest_name,CHECK_OUT_DATE,CHECK_IN_DATE
FROM   Booking_M,Guest
WHERE  Booking_M.Guest_ID = Guest.Guest_ID
and    Booking_M.guest_id not in (select BOOKING_M.guest_id
                                 from   CHECK_IN,BOOKING_M
                                  where  CHECK_IN.BOOKING_M_ID = BOOKING_M.BOOKING_M_ID)
order by guest_id
