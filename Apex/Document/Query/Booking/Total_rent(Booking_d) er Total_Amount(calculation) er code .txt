(function($) 
{
 function update(model) 
 {
 var salKey = model.getFieldKey("TOTAL_RENT"),
 total = 0;
 console.log(">> starting sum TOTAL_RENT column")
 model.forEach(function(record, index, id) 
 {
 var sal = parseFloat(record[salKey]), 
 meta = model.getRecordMetadata(id);
 if (!isNaN(sal) && !meta.deleted && !meta.agg) 
 {
 total += sal;
 }
 });
 console.log(">> setting sum SAL column to " + total)
 $s("P11_TOTAL_AMOUNT", total);
 }

 {
 sid = model.subscribe( 
 {
 onChange: function(type, change) 
{
 console.log(">> model changed ", type, change);
 if ( type === "set" ) 
 {
 
 if (change.field === "TOTAL_RENT" ) 
 {
 update( model );
 }
 } else if (type !== "move" && type !== "metaChange") 
 {
 
 update( model );
 }
 },
 progressView: $("#P11_TOTAL_AMOUNT") 
 } );
