


begin
 :TOTAL_RENT := NVL(:P11_TOTAL_BOOKED_DAYS, 0) * NVL(:ROOM_RENT,0) ;
 :P11_GRANT_TOTAL := NVL(:P11_TOTAL_AMOUNT, 0) - NVL(((:P11_TOTAL_AMOUNT*:P11_DISCOUNT)/100),0);
 :P11_DUE:=(nvl(:P11_GRANT_TOTAL,0)-nvl(:P11_ADVANCED_PAID,0));
	if :P11_GRANT_TOTAL = :P11_ADVANCED_PAID or :P11_GRANT_TOTAL < :P11_ADVANCED_PAID then
		:P11_PAYMENT_STATUS:='Paid';
	else
			:P11_PAYMENT_STATUS:='Due';
	end if;	   
end;
